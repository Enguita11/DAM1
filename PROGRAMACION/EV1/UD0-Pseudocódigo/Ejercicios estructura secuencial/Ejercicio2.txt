Proceso CalcularPerimetroYArea
	Definir base y altura como reales;
	
	Escribir "Cual es la base del rectángulo?";
	Leer base;

	Escribir "Cual es la altura del rectángulo";
	Leer altura;

	Escribir "El área del rectángulo es de: ", base * altura;
	Escribir "El perímetro del rectángulo es de: ", (2 * altura) + (2 * base);
FinProceso
