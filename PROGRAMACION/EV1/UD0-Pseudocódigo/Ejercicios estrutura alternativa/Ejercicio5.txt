Proceso NombreYContraseña
	Definir nombre, contraseña como cadena;

	Escribir "Usuario: ";
	Leer usuario;

	Escribir "Contraseña";
	Leer contraseña;

	Si nombre == "pepe" & contraseña == "asdasd" Entonces
		Escribir "Has entrado al sistema";
	Sino
		Escribir "Usuario o contraseña no válidos";
	FinSi
	
	
FinProceso

Proceso UsuarioYContraseña
	Definir usuario, contraseña como cadena;
		
	Escribir "Usuario: ";
	Leer usuario;
	Escribir "Contraseña: ";
	Leer contraseña;

	Mientras usuario != "pepe" || contraseña != "asdasd" Hacer
		Escribir "Datos erroneos";

		Escribir "Usuario: ";
		Leer usuario;
		Escribir "Contraseña: ";
		Leer contraseña;
	FinMientras
	Escribir "Has entrado al sistema";
FinProceso
