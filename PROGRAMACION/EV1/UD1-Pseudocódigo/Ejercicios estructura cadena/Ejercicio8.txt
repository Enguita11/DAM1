Proceso MayusculasAMinusuculasYViceversa

	Definir cadena, newcad como cadena;
	Definir pos como entero;
	newcad <-"";
	Escribir "Dime tu frase";
	Leer cadena;
	
	
	Para pos<-0 Hasta Longitud(cadena)-1 Con Paso 1 Hacer 
		Si (Subcadena(cad,pos,pos)= Mayúsculas(Subcadena,pos,pos)) Entonces
			//Mayúsculas
			concatenar(newcad,Minuscula(Subcadena(cad,pos,pos)))
		Sino Si (Subcadena(cad,pos,pos)= Minuscula(Subcadena,pos,pos)) Entonces
			//Minusculas
			concatenar(newcad,Minuscula(Subcadena(cad,pos,pos)))
		Sino
			//Carácter distinto
			concatenar(newcad,Subcadena(cad,pos,pos));
		FinSi
	FinPara 
	Escribir "La cadena convertida es:", newcad;
FinProceso
