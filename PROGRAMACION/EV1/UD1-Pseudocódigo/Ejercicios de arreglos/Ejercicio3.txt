	Dimensión vector[5];

		Repetir 

			Escribir "Dime la nota (0/10) ", i, " : ";
			Leer num;

		vector[i] = num;
	FinPara

	notaAlta = vector[0];
	notaBaja = vector[0];
	suma = 0;
	
	Para i <- 0 hasta longitud(vector)-1 con paso 1 hacer
		suma = suma + vector[i]

		Si notaAlta < vector[i]
			notaAlta = vector[i];
		FinSi
		Si notaBaja > vector[i]
			notaBaja = vector[i];
		FinSi

		Escribir vector[i];
	FinPara
